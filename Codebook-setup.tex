% **************************************************
% Commands for Including Source Files
% (used by the list.tex generated from build.py)
% **************************************************
% \includecpp{title}{path}: C++ (.cpp)
\newcommand{\includecpp}[2]{%
  \subsection{#1}
  \lstinputlisting[language=C++]{#2}
}

% \includepy{title}{path}: Python (.py)
\newcommand{\includepy}[2]{%
  \subsection{#1}
  \lstinputlisting[language=python]{#2}
}

% \includetex{title}{path}: LaTeX (.tex)
\newcommand{\includetex}[2]{%
  \subsection{#1}
  \input{#2}
}

% Debug helper: \geometrylen{name} gives geometry's internal length \Gm@name
\newcommand{\geometrylen}[1]{\csname Gm@#1\endcsname}

% Render '-' in listings as a plain hyphen-minus (so copied code stays valid)

\lstset{
  basicstyle=\scriptsize\ttfamily,  % font size and family of the code
  numbers=left,                     % position of line numbers
  numberstyle=\tiny,                % font size of line numbers
  stepnumber=1,                     % number every line
  numbersep=5pt,                    % distance between line numbers and code
  backgroundcolor=\color{white},    % background color
  showspaces=false,                 % do not mark spaces
  showstringspaces=false,           % do not mark spaces inside strings
  showtabs=false,                   % do not mark tabs
  frame=false,                      % no frame around the code
  tabsize=2,                        % tab width in spaces
  captionpos=b,                     % caption at the bottom
  breaklines=true,                  % wrap long lines
  breakatwhitespace=false,          % allow breaks anywhere, not only at whitespace
  escapeinside={\%*}{*)},           % escape to LaTeX inside %* ... *)
  morekeywords={*},                 % extra keywords
  keywordstyle=\bfseries\color{Blue1},
  commentstyle=\itshape\color{Red4},
  stringstyle=\itshape\color{Green4},
}
